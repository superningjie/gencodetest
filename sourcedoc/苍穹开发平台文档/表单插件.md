    import { AbstractBillPlugIn } from "@cosmic/bos-core/kd/bos/bill";

    class MyPlugin extends AbstractBillPlugIn {
        registerListener(e: EventObject): void {
            super.registerListener(e);
        }
        afterCreateNewData(e: EventObject): void {
            super.afterCreateNewData(e);
        }
        afterBindData(e: EventObject): void {
            super.afterBindData(e);
        }

        }
    }
    let plugin = new MyPlugin();

### 单据新增分录行后，设置付款日期
    
    
    afterAddRow(event: AfterAddRowEventArgs): void {
        if (event.getEntryProp().getName() == "entryentity") {
            // 订单日期
            let bizDate = this.getModel().getValue("bizDate") as Date;
            // 付款周期: 15 days, 30 days, 60 days...
            let billingDays = this.getModel().getValue("billingdays") as number;
            // 计算付款日期
            let payDate = bizDate.setDate(bizDate.getDate() + billingDays);
            // 逐行填充付款日期
            for (let row of event.getRowDataEntities()) {
                this.getModel().setValue("paydate", new Date(payDate), row.getRowIndex());
            }
        }
    }

### 操作前，增加确认提示
    
    
    // Before the operation, add a confirmation prompt.
    // Prompt the user before the operation
    beforeDoOperation(args: BeforeDoOperationEventArgs): void {
        let operate = args.getSource() as FormOperate;
        if ("pay" == operate.getOperateKey()) {
            // Payment operation
    
            // Try to read the custom operation parameter: determine whether to confirm and then execute the payment operation again
            let afterConfirm = operate.getOption().getVariableValue("afterconfirm", "");
            if (afterConfirm == "") {
                // If there is no afterconfirm parameter in the custom operation parameters: it means this is the first time the payment operation is executed, and the user needs to be prompted for confirmation
    
                // Show confirmation message
                let confirmCallBacks = new ConfirmCallBackListener("pay", this.getPluginName());
                let confirmTip = "Are you sure you want to pay xxx?";
                this.getView().showConfirm(confirmTip, MessageBoxOptions.YesNo, ConfirmTypes.Default, confirmCallBacks);
    
                // Cancel this operation before confirmation
                args.setCancel(true);
            }
        }
    }
    
    // User chooses to continue the operation
    confirmCallBack(messageBoxClosedEvent: MessageBoxClosedEvent): void {
        if ("pay" == messageBoxClosedEvent.getCallBackId()) {
            // Payment confirmation
    
            if (messageBoxClosedEvent.getResult() == MessageBoxResult.Yes) {
                // Confirm to execute the payment operation
    
                // Build custom operation parameters, mark it as confirmed to avoid repeated interaction prompts
                let operateOption = OperateOption.create();
                operateOption.setVariableValue("afterconfirm", "true");
    
                // Re-execute the payment operation with custom operation parameters
                this.getView().invokeOperation("pay", operateOption);
            }
        }
    }
    
    // Operation completed
    afterDoOperation(afterDoOperationEventArgs: AfterDoOperationEventArgs): void {
        if ("pay" == afterDoOperationEventArgs.getOperateKey() && afterDoOperationEventArgs.getOperationResult() != null && afterDoOperationEventArgs.getOperationResult().isSuccess()) {
            afterDoOperationEventArgs.getOperationResult().setShowMessage(true);
        }
    }

### 监听文本框，点击选择基础资料，捕获返回值
    
    
    // Listen to the text box, click to select base data, and capture the return value.
    // Listen to the text box, click to select base data, and capture the return value.
    registerListener(e: EventObject): void {
        let textEdit = this.getView().getControl("textfield") as TextEdit;
        textEdit.addClickListener({
            beforeClick(evt: BeforeClickEvent) {
            },
    
            click(evt: EventObject) {
                let fsp = new ListShowParameter();
                fsp.setBillFormId("bd_currency");
                fsp.setFormId("bos_listf7");
                fsp.setLookUp(true);
                fsp.setMultiSelect(false);
                fsp.setShowTitle(false);
                fsp.setHasRight(true);
    
                fsp.getOpenStyle().setShowType(ShowType.Modal);
                let css = new StyleCss();
                css.setWidth("960px");
                css.setHeight("580px");
                fsp.getOpenStyle().setInlineStyleCss(css);
    
                let closeCallBack = new CloseCallBack(plugin, "textfield");
                fsp.setCloseCallBack(closeCallBack);
    
                plugin.getView().showForm(fsp);
            }
        })
    }
    
    closedCallBack(closedCallBackEvent: ClosedCallBackEvent): void {
        if (closedCallBackEvent.getActionId() == "textfield") {
            let rows = closedCallBackEvent.getReturnData() as ListSelectedRowCollection;
            if (rows != null && rows.size() > 0) {
                this.getModel().setValue("textfield", rows.get(0).getNumber());
            }
        }
    }

### 显示进度条示例
    
    
    // Progress bar example.
    beforeBindData(e: EventObject): void {
        // Interface loading, start the progress bar
        let progressBar = this.getView().getControl("progressbar") as ProgressBar;
        progressBar.start();
        progressBar.setPercent(0, "Starting...");
        this.getView().getPageCache().put("demo_progress", "0");
    }
    
    registerListener(e: EventObject): void {
        let progressBar = this.getView().getControl("progressbar") as ProgressBar;
        progressBar.addProgressListener({
            onProgress(evt: ProgressEvent) {
                // Set progress value. The progress range is 0-100, if the progress reaches or exceeds 100, the progress bar stops automatically
                let cacheProgress = plugin.getView().getPageCache().get("demo_progress");
                if (cacheProgress == null) {
                    cacheProgress = "0";
                }
                // For demonstration, progress increases by 10% each time
                let progress = parseInt(cacheProgress) + 10;
                plugin.getView().getPageCache().put("demo_progress", progress.toString());
                evt.setProgress(progress);
                if (progress >= 100) {
                    evt.setText("Progress completed");
                } else {
                    evt.setText("In progress");
                }
            }
        })
    }

### 数量或单价属性改变时，重算金额
    
    
    // Recalculate the amount when the quantity or price property changes.
    propertyChanged(e: PropertyChangedArgs): void {
        for (let change of e.getChangeSet()) {
            let qty = this.getModel().getValue("qty", change.getRowIndex()) as BigDecimal;
            let price = this.getModel().getValue("price", change.getRowIndex()) as BigDecimal;
            let amount = qty.multiply(price);
            this.getModel().setValue("amount", amount, change.getRowIndex());
        }
    }

### 初始化树数据，监听树控件，双击返回选择的节点
    
    
    // Initialize tree data, listen to tree control, and return the selected node on double-click.
    treeViewKey = "treeviewap"
    
    afterBindData(e: EventObject): void {
        this.initTreeNodes();
    }
    
    registerListener(e: EventObject): void {
        let treeView = this.getView().getControl(treeViewKey) as TreeView;
        treeView.addTreeNodeClickListener({
            treeNodeClick(evt: TreeNodeEvent) {
            },
            treeNodeDoubleClick(evt: TreeNodeEvent) {
                let nodeid = evt.getNodeId();
                plugin.getView().returnDataToParent(nodeid);
                plugin.getView().close();
            }
        })
    }
    
    initTreeNodes(): void {
        let rootNode = new TreeNode();
        rootNode.setId("0");
        rootNode.setText("All");
    
        let node = new TreeNode();
        node.setId("0.001");
        node.setText("First Node");
    
        rootNode.addChild(node);
    
        let treeView = this.getView().getControl(treeViewKey) as TreeView;
        treeView.addNode(rootNode);
    }
