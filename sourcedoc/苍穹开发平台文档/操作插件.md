### 1、插件示例

### 定义一个操作插件
    
    
    import { AbstractOperationServicePlugIn, AddValidatorsEventArgs } from "@cosmic/bos-core/kd/bos/entity/plugin";
    import { AfterOperationArgs } from "@cosmic/bos-core/kd/bos/entity/plugin/args";
    
    class MyPlugin extends AbstractOperationServicePlugIn {
        afterExecuteOperationTransaction(e: AfterOperationArgs): void {
            super.afterExecuteOperationTransaction(e);
        }
        onAddValidators(e: AddValidatorsEventArgs): void {
            super.onAddValidators(e);
        }
    }
    let plugin = new MyPlugin();
    export { plugin };

### 操作执行前，提示确认是否继续
    
    
    //定义变量
    Interaction_Sponore = "kd.bos.plugin.sample.bill.bizoperation.bizcase.ThrowInteractionExceptionOp"
    key_integerField = "integerfield1"
    // 为操作准备数据，补充需要加载的integerfield1字段
    onPreparePropertys(e: PreparePropertysEventArgs)
        e.getFieldKeys().add(this.key_integerField);
    }
    /**
     * 操作前进行校验
     */
    beforeExecuteOperationTransaction(e: BeforeOperationArgs) {
        let maxFieldValue = 0;
        for (let dataEntity of e.getValidExtDataEntities()) {
            let fieldValue = dataEntity.getValue(this.key_integerField) as number;
            if (maxFieldValue < fieldValue) {
                maxFieldValue = fieldValue;
            }
        }
        if (maxFieldValue >= 100) {
            if (!this.showInteractionMessage()) {
                e.setCancel(true);
            }
        }
    }
    /**
     * 显示确认信息
     * @returns true继续操作，false中断操作
     */
    showInteractionMessage(): boolean {
        // 校验回调是否在弹窗之后，如果已经有弹窗不需要再提示，避免死循环
        let confirmResultString = this.getOption().getVariableValue(OperateOptionConst.INTERACTIONCONFIRMRESULT, "");
        let confirmResult = InteractionConfirmResult.fromJsonString(confirmResultString);
        // 校验确认结果是否包含在中断信息内，如果包含直接返回，避免死循环
        if (confirmResult.getResults().containsKey(this.Interaction_Sponore)) {
            let result = confirmResult.getResults().get(this.Interaction_Sponore);
            if (result == MessageBoxResult.Yes.toString()) {
                // 当用户点击确定时，继续执行后续逻辑
                return true;
            } else {
                // 当用户点击取消时，结束当前操作
                return false;
            }
        }
        // 创建一个操作中断异常上下文对象
        let interactionContext = new InteractionContext();
        // 设置异常上下文信息头
        interactionContext.setSimpleMessage("The integer field value exceeds 100, user confirmation is required to save.");
        let errorInfo = new OperateErrorInfo();
        errorInfo.setMessage("The integer field value exceeds 100");
        errorInfo.setLevel(ErrorLevel.Warning);
        interactionContext.addOperateInfo(errorInfo);
        // 抛出一个中断异常退出当前操作
        throw new KDInteractionException(this.Interaction_Sponore, interactionContext);
    }

### 操作增加自定义校验器，检查单据状态
    
    
    onAddValidators(e: AddValidatorsEventArgs) {
        let validator = this.createValidator();
        e.addValidator(validator);
    }
    createValidator(): AbstractValidator {
        // 单据状态字段
        let KEY_BILLSTATUS = "billstatus";
        let validator = this.createObject(AbstractValidator, [], {
            // 将单据状态字段添加到需要校验的字段中
            preparePropertys(): KSet<any> {
                let fields = superObject(validator).preparePropertys();
                fields.add(KEY_BILLSTATUS);
                return fields;
            },
            validate(): void {
                // 定义一个行数据模型对象，方便访问单头、单体及子单体
                let rowDataModel = new RowDataModel(validator.getEntityKey(), validator.getValidateContext().getSubEntityType());
                // 校验每一行数据
                for (let rowDataEntity of validator.getDataEntities()) {
                    rowDataModel.setRowContext(rowDataEntity.getDataEntity());
                    let billstatus = rowDataModel.getValue("billstatus") as string;
                    if (billstatus == "A") {
                        // 当单据状态是‘A’时，添加到校验异常信息
                        let info = new ValidationErrorInfo("",
                            rowDataEntity.getBillPkId(),
                            rowDataEntity.getDataEntityIndex(),
                            rowDataEntity.getRowIndex(),
                            "MyValuedate_001",
                            validator.getValidateContext().getOperateName(),
                            rowDataEntity.getBillNo() + " cannot perform this operation while in draft status. Please submit first.",
                            ErrorLevel.Error);
                        validator.getValidateResult().addErrorInfo(info);
                    }
                }
            }
        })
        return validator;
    }
